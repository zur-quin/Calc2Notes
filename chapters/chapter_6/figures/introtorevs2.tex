\documentclass[convert={svg,command=\unexpanded{{inkscape\space --pdf-poppler\space --export-type="svg"\space --pages=1\space -o\space \outfile\space \infile}}}]{standalone}% 'crop' is the default for v1.0, before it was 'preview'

\include{../../plot_settings.tex}

\begin{document}
\begin{tikzpicture}[
		area plot/.style={
				fill opacity=0.4,
				draw=black,
				fill=orange,
				mark=none,
				draw opacity = 0.15,
			} % sets up the orange slice fill in stuff
	]
	\begin{axis}[
			view = {37}{45},
			threeDaxiswithyzswap,
			xtick=4, % "generic" type problem so scale on x and y doesn't matter, remove ticks for clarity
			xmin = 0,
			xmax = 6,
			ymin = -3, % there is no y axis, the "zmax" is -ymin here, so a max third coord value of 3 put -3 here
			ymax = 3, % there is no y axis, this is -zmin
			zmin = -3.1, % actual y min on plot
			zmax = 3.1, % actual y max on plot
		]
		\tikzmath {
			% set the number of rectangles (it's off by 1 but I don't care)
			\N = 10;
		}
		\addplot3 [black, domain= 0:4,thick,samples y=0, samples = 500] (x,0,{-x^0.5}) node [midway, pin={[pin distance=.5cm, pin edge={<-,black, out=180, in=60}]below left:$y=-\sqrt{x}$}] {};
		%draw the dots that will intersect the circles later (so it can be under them)
		\addplot3 [domain= 0:4,samples y=0, samples = \N+1,mark=*,draw = none,mark size = .7pt, clip mode = individual] (x,0,{-x^0.5});

		% because of z-y axis swap, we need to use the -z axis!!!!!!!!!!!!!! or rather the second coords need an extra -!!!!!!!!!!

		%set up the loop
		\pgfplotsinvokeforeach{0,...,\N} {
			\tikzmath{
				% calculate the current x value for each iteration (otherwise it acts weird) (the 5 is hard coded the width of theinterval and "#1" is like "i" in a C loop, it is "this iteration number")
				\currentx{#1} = 4/(\N)*#1;
				% cacluctate the current radius value for each iteration (we use it multiple times so only calculate it once here)
				\currentr{#1} = \currentx{#1}^(0.5);
			}
			\begin{scope}[canvas is yz plane at x = \currentx{#1}]
				\fill[area plot] (0,0) circle (\currentr{#1}cm);
			\end{scope}
		}

		%stuff to appear on top of circles plotted after it
		\addplot3 [black, domain= 0:4,thick,samples y=0, samples = 500] (x,0,{x^0.5}) node [midway, pin={[pin distance=.5cm, pin edge={<-,black, out=145, in=-140}]above:$y=\sqrt{x}$}] {};
		% put dots 
		\addplot3 [domain= 0:4,samples y=0, samples = \N+1,mark=*,draw = none,mark size = .7pt] (x,0,{x^0.5});
		% \addplot3 [black,name path=B,thick] coordinates {(4,0,-2) (4,0,2)};

		% simulate the later bit of the x axis coming out of the middle of the last circle:
		\addplot3 [black] coordinates {(4,0,0) (5,0,0)};


		%add a lil "whoop" ya know
		\begin{scope}[canvas is yz plane at x = 5.3]
			\draw[->]  (0,.5) arc[start angle = 90, end angle = 370, radius = .5,];
		\end{scope}

	\end{axis}
\end{tikzpicture}
\end{document}
