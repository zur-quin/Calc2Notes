\documentclass[convert={svg,command=\unexpanded{{inkscape\space --pdf-poppler\space --export-type="svg"\space --pages=1\space -o\space \outfile\space \infile}}}]{standalone}% 'crop' is the default for v1.0, before it was 'preview'

\include{../../plot_settings.tex}

\begin{document}
\begin{tikzpicture}[
		declare function = {
				fun(\x) = (\x^(2))^(1/3);
				funR(\x) = 6-fun(\x);},
		area plot/.style={
				fill opacity=0.4,
				% draw=black,
				fill=orange,
				mark=none,
				draw opacity = 0.15,
			} % sets up the orange slice fill in stuff
	]
	\begin{axis}[
			twoDaxis,
			xmin = 0,
			xmax = 9.9,
			ymax = 12.5,
		]
		\addplot [black,->, domain= 0:9,thick, samples = 500, name path = func] {x^(2/3)} node [pos=0.2, pin={[pin distance=.35cm, pin edge={<-,black, out=120, in=235}]above:$y=x^{2/3}$}] {};
		\addplot [black,thick] coordinates {(8,0) (8,4)} {};
		\addplot [black, name path = floor, domain = 0:8] {0};
		\addplot [black,thick,dashed] coordinates {(8,8) (8,12)} {};
		\addplot [black,thick,dashed, name path = ceil] coordinates {(0,12) (8,12)} {};
		\addplot [black,->] coordinates {(0,6) (8,6)} node [pos=1, right] {$y=6$};
		\addplot [black,->, domain= 0:9,thick, samples = 500,dashed, name path = reflec] {-x^(2/3)+12} node [pos=0.3, pin={[pin distance=.35cm, pin edge={<-,black, out=210, in=120,solid}]below:$y=x^{2/3}$ reflected}] {};
		\diskslicevert[black,thick,r=$r$](6,6)(0.35:{funR(6)});
		\diskslicevertflip[black,thick,r=$R$](6,6)(0.6:6);
		\addplot[area plot] fill between[of = ceil and reflec, soft clip = {domain = 0:8}];
		\addplot[area plot] fill between[of = func and floor, soft clip = {domain = 0:8}];
	\end{axis}
\end{tikzpicture}
\end{document}
